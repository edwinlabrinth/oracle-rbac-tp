\section{Privilèges et rôles (Chapitre 4)}

\subsection{4.1 Création des rôles et matrice RBAC}
\textbf{Objectif.} Créer les 9 rôles de la matrice, construire la hiérarchie et attribuer les rôles aux comptes. Déléguer \code{R\_INFIRMIER} à \code{DR\_BENALI} avec \code{ADMIN OPTION}.

\shots{chu_4_1} \shots{chu_4_3}

\textbf{Observations et réponses.}
\begin{enumerate}
  \item Les 9 rôles sont créés : \code{R\_CHU\_CONNEXION}, \code{R\_SECRETAIRE}, \code{R\_INFIRMIER}, \code{R\_MEDECIN}, \code{R\_PHARMACIEN}, \code{R\_COMPTABLE}, \code{R\_DIRECTION}, \code{R\_AUDIT}, \code{R\_APPLI}.
  \item Hiérarchie : \code{R\_MEDECIN} hérite de \code{R\_INFIRMIER} qui hérite de \code{R\_CHU\_CONNEXION}. Tous les rôles héritent de \code{R\_CHU\_CONNEXION} (sauf \code{R\_AUDIT} et \code{R\_APPLI} qui ont leur propre connexion).
  \item \code{GRANT r\_infirmier TO dr\_benali WITH ADMIN OPTION} : DIR\_BENALI peut déléguer le rôle d'infirmier.
  \item Aucun privilège système n'est accordé directement aux comptes métier (conforme à E2).
\end{enumerate}
\ecart{Aucun.}
